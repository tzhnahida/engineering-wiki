# FreeRTOS 中断管理

## 从ISR API 与延迟中断处理

硬件中断应尽可能短。FreeRTOS 的做法是将耗时处理从 ISR 移到任务。

### 核心原则

```
ISR 中（数微秒）：清理中断标志 → 发信号量/通知 → 返回
  ↓ (如果唤醒了更高优先级任务，ISR 直接返回给该任务)
任务中：执行实际的耗时处理
```

### 从ISR API

所有可在 ISR 中安全调用的 FreeRTOS API 以 `FromISR` 结尾：

```c
xSemaphoreGiveFromISR()
xQueueSendFromISR()
xTaskNotifyFromISR()
xTimerStartFromISR()
// ...
```

它们内部使用中断安全的临界区宏（`portSET_INTERRUPT_MASK_FROM_ISR` / `portCLEAR_INTERRUPT_MASK_FROM_ISR`），调用者不需要额外包临界区。

**xHigherPriorityTaskWoken 机制**：

```c
void UART_IRQHandler(void) {
    BaseType_t xWoken = pdFALSE;
    uint8_t byte = UART->DR;

    xQueueSendFromISR(xRxQueue, &byte, &xWoken);
    // 如果 xWoken == pdTRUE，刚唤醒了更高优先级的任务
    portYIELD_FROM_ISR(xWoken);  // 最后调用，触发上下文切换
}
```

> 多个 FromISR 调用可用同一个 `xWoken` 变量（初始为 `pdFALSE`），最后只调用一次 `portYIELD_FROM_ISR`。

## 中断优先级配置

### Cortex-M 优先级模型

```c
// FreeRTOSConfig.h
#define configKERNEL_INTERRUPT_PRIORITY          0xF0   // SysTick/PendSV → 最低优先级
#define configMAX_SYSCALL_INTERRUPT_PRIORITY     0x50   // FromISR API 可用上限
```

**三级优先级体系（优先级数值越大 → 逻辑优先级越低）**：

| 范围 | 能否调 FreeRTOS API | 典型用途 |
|------|-------------------|---------|
| 0 ~ configMAX_SYSCALL_INTERRUPT_PRIORITY-1 | ❌ 不能调用任何 API | 时间关键型中断（如 FOC 电流环） |
| configMAX_SYSCALL_INTERRUPT_PRIORITY ~ configKERNEL_INTERRUPT_PRIORITY-1 | ✅ 可调 FromISR API | 普通外设中断 |
| configKERNEL_INTERRUPT_PRIORITY | ✅ SysTick / PendSV | 内核时钟和上下文切换 |

## 临界区实现（ARM Cortex-M4F）

BASEPRI 寄存器是 Cortex-M3/M4 **内核级中断屏蔽**机制，区别于 `cpsid i`（PRIMASK）：

- **PRIMASK（`cpsid i`）**：屏蔽所有可屏蔽中断（包括 NMI 外的所有异常）
- **BASEPRI**：屏蔽优先级**数值小于等于**设定值的中断（保留更高编号=更低优先级的中断）

FreeRTOS 使用 BASEPRI 策略，其优势在于：**高优先级中断（如 FOC 电流环）始终不受临界区影响**。

### 中断控制宏（portmacro.h）

```c
/* 关中断 — 设置 BASEPRI = configMAX_SYSCALL_INTERRUPT_PRIORITY */
#define portDISABLE_INTERRUPTS()    vPortRaiseBASEPRI()
#define portENABLE_INTERRUPTS()     vPortSetBASEPRI(0)

/* 进入/退出临界区 — C 函数，支持嵌套计数 */
#define portENTER_CRITICAL()        vPortEnterCritical()
#define portEXIT_CRITICAL()         vPortExitCritical()

/* 从 ISR 进入/退出临界区 — 保存/恢复原 BASEPRI 值 */
#define portSET_INTERRUPT_MASK_FROM_ISR()         ulPortRaiseBASEPRI()
#define portCLEAR_INTERRUPT_MASK_FROM_ISR(x)      vPortSetBASEPRI(x)
```

### BASEPRI 操作内联函数

```c
// 无条件设置 BASEPRI = configMAX_SYSCALL_INTERRUPT_PRIORITY
portFORCE_INLINE static void vPortRaiseBASEPRI( void )
{
    uint32_t ulNewBASEPRI;
    __asm volatile
    (
        " mov %0, %1                          \n"
        " msr basepri, %0                     \n"
        " isb                                 \n"
        " dsb                                 \n"
        : "=r" (ulNewBASEPRI)
        : "i" (configMAX_SYSCALL_INTERRUPT_PRIORITY)
        : "memory"
    );
}

// 保存当前 BASEPRI，然后设置
portFORCE_INLINE static uint32_t ulPortRaiseBASEPRI( void )
{
    uint32_t ulOriginalBASEPRI, ulNewBASEPRI;
    __asm volatile
    (
        " mrs %0, basepri                     \n"  // 保存原值
        " mov %1, %2                          \n"
        " msr basepri, %1                     \n"
        " isb                                 \n"
        " dsb                                 \n"
        : "=r" (ulOriginalBASEPRI), "=r" (ulNewBASEPRI)
        : "i" (configMAX_SYSCALL_INTERRUPT_PRIORITY)
        : "memory"
    );
    return ulOriginalBASEPRI;  // 返回原值，供 CLEAR 宏恢复用
}

// 恢复 BASEPRI
portFORCE_INLINE static void vPortSetBASEPRI( uint32_t ulNewMaskValue )
{
    __asm volatile
    (
        " msr basepri, %0" ::"r" (ulNewMaskValue) : "memory"
    );
}
```

### 任务级临界区（带嵌套计数）

```c
static UBaseType_t uxCriticalNesting = 0;  // 在 xPortStartScheduler 中初始化为 0

void vPortEnterCritical( void )
{
    portDISABLE_INTERRUPTS();      // 设置 BASEPRI
    uxCriticalNesting++;

    // 从任务调用的临界区，禁止在 ISR 中使用
    if( uxCriticalNesting == 1 )
        configASSERT( (portNVIC_INT_CTRL_REG & 0xFF) == 0 );  // 确保不在 ISR 中
}

void vPortExitCritical( void )
{
    configASSERT( uxCriticalNesting );
    uxCriticalNesting--;
    if( uxCriticalNesting == 0 )
        portENABLE_INTERRUPTS();   // 清除 BASEPRI（=0）
}
```

临界区可以嵌套（如在临界区内调用了含临界区的函数），只有嵌套计数降至 0 才真正使能中断。

## PendSV 与上下文切换机制

### portYIELD — 触发 PendSV

```c
#define portYIELD()                                 \
{                                                   \
    portNVIC_INT_CTRL_REG = portNVIC_PENDSVSET_BIT; \
    __asm volatile( "dsb" ::: "memory" );            \
    __asm volatile( "isb" ::: "memory" );            \
}
```

- `portNVIC_INT_CTRL_REG` 映射到 `0xE000ED04`（ICSR 寄存器）
- `portNVIC_PENDSVSET_BIT` = 位 28 — 写 1 触发 PendSV 异常
- `dsb`/`isb` 保证写操作在触发前完成

```c
// 从 ISR 触发上下文切换的等价宏
#define portYIELD_FROM_ISR( x )    do { if( x != pdFALSE ) portYIELD(); } while( 0 )
```

### 上下文切换 — xPortPendSVHandler（汇编核心）

PendSV 被配置为**最低优先级中断**，在所有其他 ISR 完成后才执行，因此上下文切换不会打断 ISR 处理。其实现（纯汇编，无栈帧）如下：

```asm
xPortPendSVHandler:
    mrs r0, psp                     ; R0 = 当前任务栈指针 PSP
    isb

    ldr r3, =pxCurrentTCB           ; R3 = &pxCurrentTCB
    ldr r2, [r3]                    ; R2 = pxCurrentTCB (当前任务 TCB)

    tst r14, #0x10                  ; 检查 EXC_RETURN 位 4:
    it eq                           ;   0 → 任务使用 FPU
    vstmdbeq r0!, {s16-s31}         ;   保存高 16 个 VFP 寄存器

    stmdb r0!, {r4-r11, r14}        ; 保存 R4~R11 + EXC_RETURN 到任务栈
    str r0, [r2]                    ; TCB->pxTopOfStack = 新 PSP

    stmdb sp!, {r0, r3}             ; 保存 R0(PSP) 和 R3(TCB 地址) 到主栈
    mov r0, #configMAX_SYSCALL_INTERRUPT_PRIORITY
    msr basepri, r0                 ; 关中断（保护调度器）
    dsb
    isb
    bl vTaskSwitchContext           ; 调用 C 函数选择下一个要运行的任务
    mov r0, #0
    msr basepri, r0                 ; 开中断
    ldmia sp!, {r0, r3}             ; 恢复 R0, R3

    ldr r1, [r3]                    ; R1 = pxCurrentTCB (新或原任务)
    ldr r0, [r1]                    ; R0 = 新任务的栈顶

    ldmia r0!, {r4-r11, r14}        ; 弹出 R4~R11 + EXC_RETURN

    tst r14, #0x10                  ; 检查 EXC_RETURN 位 4:
    it eq                           ;   0 → 新任务使用 FPU
    vldmiaeq r0!, {s16-s31}         ;   弹出高 VFP 寄存器

    msr psp, r0                     ; PSP = 新任务栈顶
    isb
    bx r14                          ; 异常返回 → 切换到新任务
```

**上下文切换保存/恢复的寄存器**：

| 谁保存 | 寄存器 | 说明 |
|--------|--------|------|
| 硬件自动（入栈） | xPSR, PC, LR, R12, R3~R0 | 中断/异常入栈时由 Cortex-M 硬件压入 |
| `stmdb r0!, {r4-r11, r14}` | R4~R11, EXC_RETURN | PendSV 手工保存 |
| `vstmdbeq r0!, {s16-s31}` | S16~S31 | 条件保存（仅 FPU 上下文） |
| **任务栈布局** | ↓ 从高地址到低地址 | 硬件帧在下，软件帧在上 |

### 首次启动 — vPortSVCHandler

`prvPortStartFirstTask` 通过 `SVC 0` 触发的 SVC 异常：

```asm
vPortSVCHandler:
    ldr r3, =pxCurrentTCB
    ldr r1, [r3]                    ; R1 = pxCurrentTCB
    ldr r0, [r1]                    ; R0 = 首个任务的栈顶
    ldmia r0!, {r4-r11, r14}        ; 弹出 R4~R11 + EXC_RETURN
    msr psp, r0                     ; PSP = 任务栈顶
    isb
    mov r0, #0
    msr basepri, r0                 ; 开中断
    bx r14                          ; 异常返回 → 进入首个任务
```

## pxPortInitialiseStack — 任务栈帧初始化

创建任务时，`xTaskCreate` 调用 port 层的 `pxPortInitialiseStack` 在任务栈上**模拟 PendSV 保存后的栈帧布局**：

```c
StackType_t * pxPortInitialiseStack(
    StackType_t * pxTopOfStack,    // 栈顶（高地址）
    TaskFunction_t pxCode,         // 任务函数入口
    void * pvParameters )          // 参数
{
    /* 模拟 PendSV + 硬件异常入栈的栈帧 */

    pxTopOfStack--;                                // 对齐
    *pxTopOfStack = portINITIAL_XPSR;              // xPSR = 0x01000000 (T 位置位)
    pxTopOfStack--;
    *pxTopOfStack = ((StackType_t)pxCode) & 0xfffffffeUL;  // PC = 任务函数（Thumb 地址）
    pxTopOfStack--;
    *pxTopOfStack = (StackType_t)prvTaskExitError;         // LR = 异常返回地址
                                                             // （防止任务函数 return）

    pxTopOfStack -= 5;                             // R12, R3, R2, R1
    *pxTopOfStack = (StackType_t)pvParameters;     // R0 = 任务参数

    pxTopOfStack--;
    *pxTopOfStack = portINITIAL_EXC_RETURN;        // EXC_RETURN = 0xfffffffd
                                                    //  位 4=0 → 使用 FPU
                                                    //  位 3=1 → 返回线程模式
                                                    //  位 2=1 → 使用 PSP
    pxTopOfStack -= 8;                             // R11, R10, R9, R8, R7, R6, R5, R4

    return pxTopOfStack;                           // 返回新的栈顶（低地址）
}
```

**栈帧布局（从高地址到低地址）**：

```
高地址 ← PSP 初始值
  xPSR = 0x01000000         (硬件帧，异常返回时自动弹出)
  PC   = task_function
  LR   = prvTaskExitError   (任务函数若 return 会进入此错误处理)
  R12  = 0
  R3   = 0
  R2   = 0
  R1   = 0
  R0   = pvParameters
  EXC_RETURN = 0xfffffffd   (软件帧，PendSV 弹出)
  R11  = 0
  R10  = 0
  R9   = 0
  R8   = 0
  R7   = 0
  R6   = 0
  R5   = 0
  R4   = 0
低地址 ← 新 PSP（pxPortInitialiseStack 返回值）
```

`portINITIAL_EXC_RETURN = 0xfffffffd` 的含义：

| 位 | 值 | 含义 |
|----|-----|------|
| 4 | 0 | FPU 已启用，进入任务时自动恢复浮点上下文 |
| 3 | 1 | 返回线程模式（非 handler 模式） |
| 2 | 1 | 使用 PSP（进程栈指针），非 MSP（主栈指针） |

## SysTick 定时器设置

### vPortSetupTimerInterrupt

`xPortStartScheduler()` 调用此函数（标记为 `__attribute__((weak))`，可被应用覆盖）：

```c
__attribute__((weak)) void vPortSetupTimerInterrupt( void )
{
    // 计算重装载值
    uint32_t ulReload = configSYSTICK_CLOCK_HZ / configTICK_RATE_HZ - 1;

    // 停止并清零 SysTick
    portNVIC_SYSTICK_CTRL_REG = 0;
    portNVIC_SYSTICK_CURRENT_VALUE_REG = 0;

    // 设置重装载
    portNVIC_SYSTICK_LOAD_REG = ulReload;
    // 启动：使用处理器时钟 + 使能中断 + 使能定时器
    portNVIC_SYSTICK_CTRL_REG = portNVIC_SYSTICK_CLK_BIT
                              | portNVIC_SYSTICK_INT_BIT
                              | portNVIC_SYSTICK_ENABLE_BIT;
}
```

相关寄存器映射：

```c
#define portNVIC_SYSTICK_CTRL_REG          (*(volatile uint32_t *)0xe000e010)
#define portNVIC_SYSTICK_LOAD_REG          (*(volatile uint32_t *)0xe000e014)
#define portNVIC_SYSTICK_CURRENT_VALUE_REG (*(volatile uint32_t *)0xe000e018)
#define portNVIC_SHPR3_REG                 (*(volatile uint32_t *)0xe000ed20)
```

PendSV 和 SysTick 被设为最低优先级（`configKERNEL_INTERRUPT_PRIORITY`）：

```c
portNVIC_SHPR3_REG |= portNVIC_PENDSV_PRI;   // PendSV 优先级
portNVIC_SHPR3_REG |= portNVIC_SYSTICK_PRI;  // SysTick 优先级
```

### xPortSysTickHandler

```c
void xPortSysTickHandler( void )
{
    portDISABLE_INTERRUPTS();        // 设置 BASEPRI 保护调度器
    {
        if( xTaskIncrementTick() != pdFALSE )
        {
            // 需要上下文切换 → 暂停 PendSV
            portNVIC_INT_CTRL_REG = portNVIC_PENDSVSET_BIT;
        }
    }
    portENABLE_INTERRUPTS();         // 清除 BASEPRI
}
```

SysTick 不会直接进行上下文切换，而是设置 PendSV 位。PendSV 在所有其他 ISR 执行完毕后（因其最低优先级）运行真正的切换，保证了上下文切换不会阻塞实时中断。

## VFP（浮点单元）管理

CM4F 内核带有 FPU。FreeRTOS 支持**懒加载（lazy stacking）** 来减少每次上下文切换时保存/恢复 VFP 寄存器的开销。

**启用 FPU**（在 `xPortStartScheduler` 中）：

```c
static void vPortEnableVFP( void )
{
    __asm volatile
    (
        " ldr.w r0, =0xE000ED88        \n"  // CPACR 寄存器地址
        " ldr r1, [r0]                 \n"
        " orr r1, r1, #(0xf << 20)     \n"  // 使能 CP10 和 CP11
        " str r1, [r0]                 \n"
        " bx r14                       \n"
    );
}

// 设置 FPCCR 使能懒加载
*(portFPCCR) |= portASPEN_AND_LSPEN_BITS;  // ASPEN=1, LSPEN=1
```

**懒加载的含义**：只有任务实际使用了浮点指令，才在上下文切换中保存 VFP 寄存器。`EXC_RETURN` 的位 4 指示当前任务是否使用 FPU。

PendSV 中的 FPU 处理：
```asm
tst r14, #0x10        ; EXC_RETURN 位 4 = 0 → 任务用过 FPU
it eq
vstmdbeq r0!, {s16-s31}   ; 保存 S16~S31（高 16 个 VFP 寄存器）
; S0~S15 由硬件懒加载自动管理
```

## 中断优先级验证机制

`portASSERT_IF_INTERRUPT_PRIORITY_INVALID()` 是一个运行时断言，检查当前 ISR 是否违反了 `configMAX_SYSCALL_INTERRUPT_PRIORITY` 约束：

```c
void vPortValidateInterruptPriority( void )
{
    uint32_t ulCurrentInterrupt;
    __asm volatile( "mrs %0, ipsr" : "=r" (ulCurrentInterrupt) );

    if( ulCurrentInterrupt >= 16 )  // 用户中断号
    {
        // 查优先级寄存器
        uint8_t ucPriority = pcInterruptPriorityRegisters[ulCurrentInterrupt];
        // 必须 ≥ configMAX_SYSCALL_INTERRUPT_PRIORITY
        configASSERT( ucPriority >= ucMaxSysCallPriority );
    }
}
```

此函数在 `xPortIsInsideInterrupt()` 和优先级分组校验的配合下，确保：
1. 高优先级（数值小）中断不会误调 FreeRTOS API
2. NVIC 优先级分组未设置为子优先级模式（所有优先级位必须是抢占位）

## 其他端口层接口

所有 FreeRTOS 移植必须实现的接口（定义于 `portable.h`）：

| 函数/宏 | 说明 |
|---------|------|
| `pxPortInitialiseStack()` | 初始化任务栈帧，见表上节 |
| `xPortStartScheduler()` | 配置 SysTick，设 PendSV/SysTick 优先级，使能 FPU，调用 `prvPortStartFirstTask()` |
| `vPortEndScheduler()` | 空函数（Cortex-M 无法退出调度器）|
| `portYIELD()` | 设置 PendSV 位触发上下文切换 |
| `portYIELD_FROM_ISR(x)` | ISR 中条件触发 PendSV |
| `portENTER_CRITICAL()` / `portEXIT_CRITICAL()` | 任务级临界区（BASEPRI + 嵌套） |
| `portSET_INTERRUPT_MASK_FROM_ISR()` / `portCLEAR_INTERRUPT_MASK_FROM_ISR()` | ISR 级临界区（保存/恢复 BASEPRI） |
| `portDISABLE_INTERRUPTS()` / `portENABLE_INTERRUPTS()` | 基础中断屏蔽/使能 |
| `portTICK_TYPE_IS_ATOMIC` | 1 → 32 位平台，读 tick 不需要临界区保护 |
